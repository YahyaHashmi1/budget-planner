# --- Useful commands---
# Start Docker containers: docker-compose up -d
# Stop Docker containers: docker-compose down
# Delete Docker containers: docker-compose down -v

# Run Postgres commands in Docker: docker exec -it paycheck-tracker-db psql -U paycheck_user -d paycheck_tracker
# List tables: \dt
# List columns: \d table_name
# View all data in a table: SELECT * FROM table_name;

# Generate migration: alembic revision --autogenerate -m "migration message"
# Push migration: alembic upgrade head
# Display current migration: alembic current
# Point to previous migration: alembic downgrade -1
# Discard current migration and point to specific migration: alembic stamp --purge <revision_id>

# Start FastAPI server: python -m uvicorn app.main:app --reload
